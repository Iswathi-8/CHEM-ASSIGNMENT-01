def safe_name(s):
    return re.sub(r'[^A-Za-z0-9_-]+','_',s or 'molecule').strip('_')[:60] or 'molecule'

def get_molecule(key):
    return dict(MOLECULES[key])

@app.context_processor
def globals_for_templates():
    return {'molecule_catalog': MOLECULES}

@app.route('/visualize', methods=['GET','POST'])
def visualize():
    result = None
    if request.method == 'POST':
        key = request.form.get('molecule','ethanol')
        if key not in MOLECULES:
            flash('Choose one of the supported reference molecules.', 'error')
        else:
            m = get_molecule(key)
            # Without a chemistry engine, save a SMILES text file instead of claiming to create a true 3D structure.
            filename = safe_name(m['name']) + '.smi'
            with open(os.path.join(STRUCTURES_DIR, filename),'w',encoding='utf-8') as f:
                f.write(m['smiles'] + '\t' + m['name'] + '\n')
            result = {'molecule':m, 'file':filename}
            flash('Reference molecule loaded. Download the SMILES file to open/import it in compatible chemistry software.', 'success')
    return render_template('visualize.html', title='Molecular Visualization', result=result)

@app.route('/properties', methods=['GET','POST'])
def properties():
    result = None
    if request.method == 'POST':
        key = request.form.get('molecule','ethanol')
        if key not in MOLECULES:
            flash('Choose a molecule from the reference library.', 'error')
        else:
            result = get_molecule(key)
            result['value_type'] = 'Preset educational reference data'
            path = os.path.join(REPORTS_DIR, safe_name(result['name']) + '_properties.csv')
            with open(path,'w',newline='',encoding='utf-8') as f:
                writer=csv.writer(f); writer.writerow(['Property','Value'])
                for k,v in result.items():
                    if k not in ('note','value_type'): writer.writerow([k,v])
            flash('Reference properties shown and CSV exported.', 'success')
    return render_template('properties.html', title='Molecular Properties', result=result)

@app.route('/analysis', methods=['GET','POST'])
def analysis():
    rows=None; graph=None; property_key='molecular_weight'
    if request.method=='POST':
        keys=request.form.getlist('molecules')
        property_key=request.form.get('property','molecular_weight')
        allowed={'molecular_weight':'Molecular weight (g/mol)','logp':'LogP','tpsa':'TPSA (Å²)','hbd':'H-bond donors','hba':'H-bond acceptors','rings':'Ring count'}
        if property_key not in allowed: property_key='molecular_weight'
        unique=[]
        for k in keys:
            if k in MOLECULES and k not in unique: unique.append(k)
        if len(unique)<2:
            flash('Select at least two molecules to compare.', 'error')
        else:
            rows=[get_molecule(k) for k in unique]
            path=os.path.join(REPORTS_DIR,'structure_property_comparison.csv')
            with open(path,'w',newline='',encoding='utf-8') as f:
                writer=csv.DictWriter(f,fieldnames=list(rows[0].keys())); writer.writeheader(); writer.writerows(rows)
            vals=[float(r[property_key]) for r in rows]; maxv=max(vals) if max(vals)>0 else 1
            bars=[]
            for r,v in zip(rows,vals):
                width=max(2,int((v/maxv)*100))
                bars.append({'name':r['name'],'value':v,'width':width})
            graph={'label':allowed[property_key],'bars':bars}
            flash('Comparison and CSV generated from preset reference values.', 'success')
    return render_template('analysis.html', title='Structure–Property Analysis', rows=rows, graph=graph, property_key=property_key)

@app.route('/assessment', methods=['GET','POST'])
def assessment():
    score=None; review=None; student=None; saved_file=None; stage='identity'
    if request.method == 'POST':
        action=request.form.get('action','confirm')
        username=request.form.get('username','').strip()
        regno=request.form.get('regno','').strip()
        if action in ('confirm','begin','edit'):
            if not username or not regno:
                flash('Please enter both student name and registration number.', 'error')
                stage='identity'
            elif action == 'confirm':
                student={'username':username,'regno':regno}
                stage='confirm'
            elif action == 'edit':
                student={'username':username,'regno':regno}
                stage='identity'
            else:
                student={'username':username,'regno':regno}
                stage='quiz'
        elif action == 'submit':
            if not username or not regno:
                flash('Student identity is missing. Please start again.', 'error')
                stage='identity'
            else:
                review=[]; score=0
                for i,q in enumerate(QUIZ):
                    try: selected=int(request.form.get('q'+str(i),'-1'))
                    except ValueError: selected=-1
                    correct=selected==q['answer']
                    if correct: score+=1
                    review.append({'question':q['q'],'selected':q['options'][selected] if 0<=selected<len(q['options']) else 'Not answered','correct_answer':q['options'][q['answer']],'correct':correct})
                student={'username':username,'regno':regno}
                payload={'timestamp':datetime.now().isoformat(),'username':username,'registration_number':regno,'score':score,'total':len(QUIZ),'review':review}
                saved_file='assessment_'+safe_name(regno)+'_'+datetime.now().strftime('%Y%m%d_%H%M%S')+'.json'
                with open(os.path.join(REPORTS_DIR,saved_file),'w',encoding='utf-8') as f: json.dump(payload,f,indent=2)
                csvpath=os.path.join(REPORTS_DIR,'assessment_results.csv')
                exists=os.path.exists(csvpath)
                with open(csvpath,'a',newline='',encoding='utf-8') as f:
                    w=csv.writer(f)
                    if not exists: w.writerow(['timestamp','username','registration_number','score','total'])
                    w.writerow([payload['timestamp'],username,regno,score,len(QUIZ)])
                flash('Assessment submitted. Student identity and score have been saved locally.', 'success')
                stage='result'
    return render_template('assessment.html', title='Assessment', quiz=QUIZ, score=score, review=review, student=student, saved_file=saved_file, stage=stage)

@app.route('/download/<path:filename>')
def download(filename):
    for folder in (STRUCTURES_DIR,REPORTS_DIR,GRAPHS_DIR):
        candidate=os.path.abspath(os.path.join(folder,filename))
        if candidate.startswith(os.path.abspath(folder)+os.sep) and os.path.isfile(candidate):
            return send_from_directory(folder,filename,as_attachment=True)
    abort(404)
